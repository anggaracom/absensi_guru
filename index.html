Panduan Panduan Backend Google Sheets (Salin Skrip Ini)

Buka Google Sheets baru -> Klik menu Ekstensi -> Apps Script -> Hapus kode lama dan tempel kode berikut lalu klik Deploy -> Web App (Anyone):

// ==================== KODE GOOGLE APPS SCRIPT (GAS) ====================
function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action == 'getData') {
    var users = getSheetData(ss.getSheetByName('Users'));
    var attendance = getSheetData(ss.getSheetByName('Attendance'));
    var settings = getSheetData(ss.getSheetByName('Settings'));
    return responseJSON({ users: users, attendance: attendance, settings: settings });
  }
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (data.action == 'absen') {
    var sheet = ss.getSheetByName('Attendance');
    sheet.appendRow([data.id, data.nip, data.nama, data.tanggal, data.jamMasuk, data.jamPulang, data.jarak, data.status]);
    return responseJSON({ status: 'success' });
  }
  return responseJSON({ status: 'error' });
}

function getSheetData(sheet) {
  if(!sheet) return [];
  var rows = sheet.getDataRange().getValues();
  var headers = rows[0];
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    data.push(obj);
  }
  return data;
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
